# 1.0.4 (the K790 build from speedrun.com): the air run on Pine Island: a tiger jump off a roof edge, its landing roll goes on in the air, the run it turns into is still in the air, UP turns the fall into speed
set jar packages/content/playman/original/PER_K790_1.0.4.jar
set frame-method s
set unlock true
set layout side
set scale 1
set zoom 3
set pad 0
set frame 1920x1080
boot 2
wait 2.5
press soft2      # the "turn sound on?" prompt
wait 2.8
press select     # skip the publisher's splash
wait 2.6
press select     # main menu: single game
wait 2.2
press select     # character: Playman
wait 2.2
repeat 11 0.6 right   # levels 2 … 12
wait 1.0
press select     # the level
wait 2.0
repeat 2 0.6 right    # missions: sprint, flag hunt
wait 0.8
press select     # the briefing
wait 2.5
press select     # go: a flag hunt has no flyover, this press starts the race
route packages/tools/tas-out/routes/air-speed-11-flags.json 1   # the jar reads a route one frame later than the remake
tail 5
